// Check missing and duplicate IDs

WITH ids AS (
    SELECT 'PATIENTS' AS table_name,
           NULLIF(TRIM(patient_id), '') AS id
    FROM RAW.PATIENTS

    UNION ALL
    SELECT 'PROVIDERS', NULLIF(TRIM(provider_id), '')
    FROM RAW.PROVIDERS

    UNION ALL
    SELECT 'CLAIMS', NULLIF(TRIM(claim_id), '')
    FROM RAW.CLAIMS

    UNION ALL
    SELECT 'DIAGNOSES_HISTORY', NULLIF(TRIM(record_id), '')
    FROM RAW.DIAGNOSES_HISTORY
)
SELECT
    table_name,
    COUNT(*) AS total_rows,
    COUNT(*) - COUNT(id) AS missing_ids,
    COUNT(id) - COUNT(DISTINCT id) AS duplicate_extra_rows
FROM ids
GROUP BY table_name
ORDER BY table_name;

// Check Date conversions

WITH date_checks AS (
    SELECT 'PATIENTS.date_of_birth' AS field,
           date_of_birth AS date_value, 'YYYY-MM-DD' AS date_format
    FROM RAW.PATIENTS
    UNION ALL
    SELECT 'PATIENTS.registration_date', registration_date, 'YYYY-MM-DD'
    FROM RAW.PATIENTS
    UNION ALL
    SELECT 'CLAIMS.claim_date', claim_date, 'DD/MM/YYYY'
    FROM RAW.CLAIMS
    UNION ALL
    SELECT 'CLAIMS.payment_date', payment_date, 'DD/MM/YYYY'
    FROM RAW.CLAIMS
    UNION ALL
    SELECT 'DIAGNOSES_HISTORY.diagnosis_date', diagnosis_date, 'YYYY-MM-DD'
    FROM RAW.DIAGNOSES_HISTORY
    UNION ALL
    SELECT 'DIAGNOSES_HISTORY.follow_up_date', follow_up_date, 'YYYY-MM-DD'
    FROM RAW.DIAGNOSES_HISTORY
)
SELECT
    field,
    COUNT(*) AS invalid_dates
FROM date_checks
WHERE NULLIF(TRIM(date_value), '') IS NOT NULL
  AND TRY_TO_DATE(TRIM(date_value), date_format) IS NULL
GROUP BY field;


//Check whether every claim and diagnosis references an existing patient and provider

SELECT 'CLAIMS → PATIENTS' AS relationship,
       COUNT(*) AS unmatched_rows
FROM RAW.CLAIMS c
WHERE NOT EXISTS (
    SELECT 1 FROM RAW.PATIENTS p
    WHERE TRIM(p.patient_id) = TRIM(c.patient_id)
)

UNION ALL

SELECT 'CLAIMS → PROVIDERS', COUNT(*)
FROM RAW.CLAIMS c
WHERE NOT EXISTS (
    SELECT 1 FROM RAW.PROVIDERS p
    WHERE TRIM(p.provider_id) = TRIM(c.provider_id)
)

UNION ALL

SELECT 'DIAGNOSES → PATIENTS', COUNT(*)
FROM RAW.DIAGNOSES_HISTORY d
WHERE NOT EXISTS (
    SELECT 1 FROM RAW.PATIENTS p
    WHERE TRIM(p.patient_id) = TRIM(d.patient_id)
)

UNION ALL

SELECT 'DIAGNOSES → PROVIDERS', COUNT(*)
FROM RAW.DIAGNOSES_HISTORY d
WHERE NOT EXISTS (
    SELECT 1 FROM RAW.PROVIDERS p
    WHERE TRIM(p.provider_id) = TRIM(d.provider_id)
);

// Check numeric and boolean fields

WITH checks AS (
    SELECT 'PROVIDERS.years_experience' AS field,
           years_experience AS value, 'INTEGER' AS expected_type
    FROM RAW.PROVIDERS
    UNION ALL
    SELECT 'PROVIDERS.consultation_fee', consultation_fee, 'AMOUNT'
    FROM RAW.PROVIDERS
    UNION ALL
    SELECT 'CLAIMS.billed_amount', billed_amount, 'AMOUNT'
    FROM RAW.CLAIMS
    UNION ALL
    SELECT 'CLAIMS.approved_amount', approved_amount, 'AMOUNT'
    FROM RAW.CLAIMS
    UNION ALL
    SELECT 'CLAIMS.patient_copay', patient_copay, 'AMOUNT'
    FROM RAW.CLAIMS
    UNION ALL
    SELECT 'PATIENTS.is_active', is_active, 'BOOLEAN'
    FROM RAW.PATIENTS
    UNION ALL
    SELECT 'PROVIDERS.is_active', is_active, 'BOOLEAN'
    FROM RAW.PROVIDERS
    UNION ALL
    SELECT 'DIAGNOSES_HISTORY.is_chronic', is_chronic, 'BOOLEAN'
    FROM RAW.DIAGNOSES_HISTORY
)
SELECT field, COUNT(*) AS invalid_or_missing_rows
FROM checks
WHERE NULLIF(TRIM(value), '') IS NULL
   OR (
       expected_type = 'INTEGER'
       AND (
           NOT REGEXP_LIKE(TRIM(value), '^[0-9]+$')
           OR TRY_TO_NUMBER(TRIM(value)) IS NULL
       )
   )
   OR (
       expected_type = 'AMOUNT'
       AND TRY_TO_DECIMAL(TRIM(value), 18, 2) IS NULL
   )
   OR (
       expected_type = 'BOOLEAN'
       AND LOWER(TRIM(value)) NOT IN ('true', 'false')
   )
GROUP BY field;

SELECT
    COUNT(*) AS total_patients,
    MIN(date_of_birth) AS earliest_birth_date,
    MAX(date_of_birth) AS latest_birth_date,
    COUNT_IF(is_active) AS active_patients
FROM CLEAN.PATIENTS;


SELECT
    COUNT(*) AS total_providers,
    MIN(years_experience) AS min_years_experience,
    MAX(years_experience) AS max_years_experience,
    MIN(consultation_fee) AS min_consultation_fee,
    MAX(consultation_fee) AS max_consultation_fee,
    COUNT_IF(is_active) AS active_providers
FROM CLEAN.PROVIDERS;


SELECT
    COUNT(*) AS total_claims,
    MIN(claim_date) AS earliest_claim_date,
    MAX(claim_date) AS latest_claim_date,
    SUM(billed_amount) AS total_billed_amount,
    SUM(approved_amount) AS total_approved_amount,
    SUM(patient_copay) AS total_patient_copay,
    COUNT_IF(payment_date IS NULL) AS missing_payment_dates,
    COUNT_IF(denial_reason IS NULL) AS missing_denial_reasons
FROM CLEAN.CLAIMS;


SELECT
    COUNT(*) AS total_diagnoses,
    MIN(diagnosis_date) AS earliest_diagnosis_date,
    MAX(diagnosis_date) AS latest_diagnosis_date,
    COUNT_IF(is_chronic) AS chronic_diagnoses,
    COUNT_IF(follow_up_date < diagnosis_date)
        AS follow_ups_before_diagnosis

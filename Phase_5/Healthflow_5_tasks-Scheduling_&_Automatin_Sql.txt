SHOW TABLES IN SCHEMA HEALTHFLOW_DB.CURATED;

SHOW TASKS IN DATABASE HEALTHFLOW_DB;


// Create the Root Task
USE DATABASE HEALTHFLOW_DB;
USE SCHEMA CURATED;
USE WAREHOUSE HF_TRANSFORM_WH;

CREATE OR REPLACE TASK healthflow_root_task
    WAREHOUSE = HF_TRANSFORM_WH
    SCHEDULE = '10 MINUTE'
AS
    SELECT CURRENT_TIMESTAMP() AS pipeline_start_time;

// Verify the Root Task
SHOW TASKS IN SCHEMA HEALTHFLOW_DB.CURATED;

//Verify the Patients Stream
USE DATABASE HEALTHFLOW_DB;
USE SCHEMA RAW;

SHOW STREAMS IN SCHEMA HEALTHFLOW_DB.RAW;

//BRANCHESck whether Patients Stream contains pending changes
SELECT SYSTEM$STREAM_HAS_DATA(
    'HEALTHFLOW_DB.RAW.PATIENTS_STREAM'


DESCRIBE TABLE HEALTHFLOW_DB.CURATED.PATIENTS;
) AS HAS_NEW_DATA;

// Create the Patients Processing Task
USE DATABASE HEALTHFLOW_DB;
USE SCHEMA CURATED;

CREATE OR REPLACE TASK PROCESS_PATIENTS_TASK
    WAREHOUSE = HF_TRANSFORM_WH
    AFTER HEALTHFLOW_ROOT_TASK
    WHEN SYSTEM$STREAM_HAS_DATA(
        'HEALTHFLOW_DB.RAW.PATIENTS_STREAM'
    )
AS
    CALL CURATED.PROCESS_PATIENTS();

SHOW TASKS IN SCHEMA HEALTHFLOW_DB.CURATED;

// Create the Claims Task

        'HEALTHFLOW_DB.RAW.CLAIMS_STREAM'
    )
AS

SHOW TASKS IN SCHEMA HEALTHFLOW_DB.CURATED;


// Create the Diagnoses Task

USE SCHEMA CURATED;

CREATE OR REPLACE TASK PROCESS_DIAGNOSES_TASK
    WAREHOUSE = HF_TRANSFORM_WH
    AFTER HEALTHFLOW_ROOT_TASK

        'HEALTHFLOW_DB.RAW.DIAGNOSES_STREAM'
    )
AS

SHOW TASKS IN SCHEMA HEALTHFLOW_DB.CURATED;


// Verify and Activate the Task DAG
SELECT
    NAME,

    SCHEDULE,
    PREDECESSORS
FROM TABLE(

);


SHOW TASKS IN SCHEMA HEALTHFLOW_DB.CURATED;


// Resume the Snowflake Tasks
USE DATABASE HEALTHFLOW_DB;
USE SCHEMA CURATED;

-- 1. Resume Patients Task
ALTER TASK PROCESS_PATIENTS_TASK RESUME;

-- 2. Resume Claims Task
ALTER TASK PROCESS_CLAIMS_TASK RESUME;

-- 3. Resume Diagnoses Task
ALTER TASK PROCESS_DIAGNOSES_TASK RESUME;

-- 4. Resume Root Task LAST
ALTER TASK HEALTHFLOW_ROOT_TASK RESUME;

///// Recover Deleted Claims Using Time Travel

// Verify the historical records
SELECT COUNT(*) AS HISTORICAL_CLAIMS
FROM HEALTHFLOW_DB.CURATED.CLAIMS
AT (TIMESTAMP => '2026-11-07 09:06:22.624 -0800'::TIMESTAMP_TZ)
WHERE DIAGNOSIS_CODE = 'I10';

// Use AT (OFFSET) for recovery
SELECT DATEDIFF(
    'second',
    '2026-11-07 09:06:22.624 -0800'::TIMESTAMP_TZ,
    CURRENT_TIMESTAMP()
) AS SECONDS_ELAPSED;

// Verify
SELECT COUNT(*) AS RECOVERABLE_CLAIMS
FROM HEALTHFLOW_DB.CURATED.CLAIMS
AT (OFFSET => -1800)
WHERE DIAGNOSIS_CODE = 'I10';

// Restore the deleted records
INSERT INTO HEALTHFLOW_DB.CURATED.CLAIMS
SELECT *
FROM HEALTHFLOW_DB.CURATED.CLAIMS
AT (OFFSET => -1800)
WHERE DIAGNOSIS_CODE = 'I10';

// Validate recovery
SELECT COUNT(*) AS RESTORED_CLAIMS
FROM HEALTHFLOW_DB.CURATED.CLAIMS
WHERE DIAGNOSIS_CODE = 'I10';

////// Warehouse Credit Consumption

// Total credits consumed by warehouse
USE WAREHOUSE HF_DEV_WH;

SELECT
    WAREHOUSE_NAME,
    ROUND(SUM(CREDITS_USED), 2) AS TOTAL_CREDITS,
    ROUND(SUM(CREDITS_USED_COMPUTE), 2) AS COMPUTE_CREDITS,
    ROUND(SUM(CREDITS_USED_CLOUD_SERVICES), 2) AS CLOUD_SERVICE_CREDITS
FROM SNOWFLAKE.ACCOUNT_USAGE.WAREHOUSE_METERING_HISTORY
WHERE START_TIME >= DATEADD(DAY, -30, CURRENT_TIMESTAMP())
GROUP BY WAREHOUSE_NAME
ORDER BY TOTAL_CREDITS DESC;

// Daily Credit Usage Trend
SELECT
    DATE_TRUNC('DAY', START_TIME)::DATE AS USAGE_DATE,
    WAREHOUSE_NAME,
    ROUND(SUM(CREDITS_USED), 2) AS DAILY_CREDITS
FROM SNOWFLAKE.ACCOUNT_USAGE.WAREHOUSE_METERING_HISTORY
WHERE START_TIME >= DATEADD(DAY, -30, CURRENT_TIMESTAMP())
GROUP BY 1, 2
ORDER BY USAGE_DATE DESC, DAILY_CREDITS DESC;

// Top 10 Most Expensive Queries
SELECT
    q.QUERY_ID,
    q.WAREHOUSE_NAME,
    q.QUERY_TYPE,
    ROUND(a.CREDITS_ATTRIBUTED_COMPUTE, 4) AS QUERY_CREDITS,
    ROUND(q.TOTAL_ELAPSED_TIME / 1000, 2) AS EXECUTION_SECONDS,
    LEFT(q.QUERY_TEXT, 120) AS QUERY_TEXT
FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_ATTRIBUTION_HISTORY a
JOIN SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY q
    ON a.QUERY_ID = q.QUERY_ID
WHERE a.START_TIME >= DATEADD(DAY, -30, CURRENT_TIMESTAMP())
ORDER BY QUERY_CREDITS DESC
LIMIT 10;

// Identify Excessive Partition Scanning
SELECT
    QUERY_ID,
    WAREHOUSE_NAME,
    PARTITIONS_SCANNED,
    PARTITIONS_TOTAL,
    ROUND(
        100.0 * PARTITIONS_SCANNED /
        NULLIF(PARTITIONS_TOTAL, 0), 2
    ) AS SCAN_PERCENT,
    ROUND(BYTES_SCANNED / POWER(1024, 2), 2) AS MB_SCANNED,
    LEFT(QUERY_TEXT, 120) AS QUERY_TEXT
FROM SNOWFLAKE.ACCOUNT_USAGE.QUERY_HISTORY
WHERE START_TIME >= DATEADD(DAY, -30, CURRENT_TIMESTAMP())
  AND PARTITIONS_TOTAL > 0
  AND PARTITIONS_SCANNED > 0
ORDER BY PARTITIONS_SCANNED DESC
LIMIT 10;

// Identify Unused Warehouses
SHOW WAREHOUSES;

WITH warehouses AS (
    SELECT "name" AS warehouse_name
    FROM TABLE(RESULT_SCAN(LAST_QUERY_ID()))
),
usage AS (
    SELECT
        WAREHOUSE_NAME,
        SUM(CREDITS_USED) AS TOTAL_CREDITS
    FROM SNOWFLAKE.ACCOUNT_USAGE.WAREHOUSE_METERING_HISTORY
    WHERE START_TIME >= DATEADD(DAY, -30, CURRENT_TIMESTAMP())
    GROUP BY WAREHOUSE_NAME
)
SELECT
    w.warehouse_name,
    ROUND(COALESCE(u.TOTAL_CREDITS, 0), 2) AS CREDITS_LAST_30_DAYS,
    CASE
        WHEN u.WAREHOUSE_NAME IS NULL THEN 'NO RECORDED USAGE'
        ELSE 'ACTIVE'
    END AS USAGE_STATUS
FROM warehouses w
LEFT JOIN usage u
    ON w.warehouse_name = u.WAREHOUSE_NAME
ORDER BY CREDITS_LAST_30_DAYS ASC;

// Storage Usage Analysis (Final Check)
SELECT
    USAGE_DATE,
    DATABASE_NAME,
    ROUND(AVERAGE_DATABASE_BYTES / POWER(1024, 3), 4)
        AS DATABASE_STORAGE_GB,
    ROUND(AVERAGE_FAILSAFE_BYTES / POWER(1024, 3), 4)
        AS FAILSAFE_STORAGE_GB
FROM SNOWFLAKE.ACCOUNT_USAGE.DATABASE_STORAGE_USAGE_HISTORY
WHERE DATABASE_NAME = 'HEALTHFLOW_DB'
  AND USAGE_DATE >= DATEADD(DAY, -30, CURRENT_DATE())
  AND DELETED IS NULL
ORDER BY USAGE_DATE DESC;

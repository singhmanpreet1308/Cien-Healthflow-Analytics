// Create a PATIENTS processing procedure
USE DATABASE HEALTHFLOW_DB;
USE WAREHOUSE HF_TRANSFORM_WH;

CREATE OR REPLACE PROCEDURE
CURATED.PROCESS_PATIENTS()
RETURNS VARCHAR
LANGUAGE SQL
AS
$$
BEGIN

    MERGE INTO CURATED.PATIENTS AS tgt

    USING (
        SELECT
            TRY_TO_NUMBER(patient_id)::NUMBER(38,0)
                AS patient_id,
            NULLIF(TRIM(first_name), '')
                AS first_name,
            NULLIF(TRIM(last_name), '')
                AS last_name,
            TRY_TO_DATE(date_of_birth)
                AS date_of_birth,
            NULLIF(TRIM(gender), '')
                AS gender,
            NULLIF(TRIM(blood_group), '')
                AS blood_group,
            NULLIF(TRIM(phone), '')
                AS phone,
            NULLIF(LOWER(TRIM(email)), '')
                AS email,
            NULLIF(TRIM(address_city), '')
                AS address_city,
            NULLIF(TRIM(address_state), '')
                AS address_state,
            NULLIF(TRIM(country), '')
                AS country,
            NULLIF(TRIM(insurance_provider), '')
                AS insurance_provider,
            NULLIF(TRIM(insurance_plan_type), '')
                AS insurance_plan_type,
            TRY_TO_DATE(registration_date)
                AS registration_date,
            TRY_TO_BOOLEAN(is_active::VARCHAR)
                AS is_active,

            DATEDIFF(
                YEAR,
                TRY_TO_DATE(date_of_birth),
                CURRENT_DATE()
            )
            - IFF(
                DATEADD(
                    YEAR,
                    DATEDIFF(
                        YEAR,
                        TRY_TO_DATE(date_of_birth),
                        CURRENT_DATE()
                    ),
                    TRY_TO_DATE(date_of_birth)
                ) > CURRENT_DATE(),
                1, 0
            ) AS patient_age

        FROM RAW.PATIENTS_STREAM

        WHERE METADATA$ACTION = 'INSERT'

        QUALIFY ROW_NUMBER() OVER (
            PARTITION BY patient_id
            ORDER BY METADATA$ROW_ID
        ) = 1

    ) AS src

    ON tgt.patient_id = src.patient_id

    WHEN MATCHED THEN UPDATE SET
        tgt.first_name = src.first_name,
        tgt.last_name = src.last_name,
        tgt.date_of_birth = src.date_of_birth,
        tgt.gender = src.gender,
        tgt.blood_group = src.blood_group,
        tgt.phone = src.phone,
        tgt.email = src.email,
        tgt.address_city = src.address_city,
        tgt.address_state = src.address_state,
        tgt.country = src.country,
        tgt.insurance_provider = src.insurance_provider,
        tgt.insurance_plan_type = src.insurance_plan_type,
        tgt.registration_date = src.registration_date,
        tgt.is_active = src.is_active,
        tgt.patient_age = src.patient_age

    WHEN NOT MATCHED THEN INSERT (
        patient_id, first_name, last_name,
        date_of_birth, gender, blood_group,
        phone, email, address_city,
        address_state, country,
        insurance_provider, insurance_plan_type,
        registration_date, is_active, patient_age
    )
    VALUES (
        src.patient_id, src.first_name, src.last_name,
        src.date_of_birth, src.gender, src.blood_group,
        src.phone, src.email, src.address_city,
        src.address_state, src.country,
        src.insurance_provider, src.insurance_plan_type,
        src.registration_date, src.is_active,
        src.patient_age
    );

    RETURN 'PATIENTS MERGE completed successfully';

END;
$$;

USE DATABASE HEALTHFLOW_DB;
USE WAREHOUSE HF_TRANSFORM_WH;

CREATE OR REPLACE PROCEDURE CURATED.PROCESS_CLAIMS()
RETURNS VARCHAR
LANGUAGE SQL
AS
$$
BEGIN

    MERGE INTO CURATED.CLAIMS AS tgt

    USING (
        SELECT
            TRY_TO_NUMBER(claim_id::VARCHAR)::NUMBER(38,0)
                AS claim_id,

            TRY_TO_NUMBER(patient_id::VARCHAR)::NUMBER(38,0)
                AS patient_id,

            TRY_TO_NUMBER(provider_id::VARCHAR)::NUMBER(38,0)
                AS provider_id,

            NULLIF(TRIM(diagnosis_code), '')
                AS diagnosis_code,

            NULLIF(TRIM(diagnosis_description), '')
                AS diagnosis_description,

            COALESCE(
                TRY_TO_DATE(claim_date, 'YYYY-MM-DD'),
                TRY_TO_DATE(claim_date, 'DD/MM/YYYY')
            ) AS claim_date,

            NULLIF(TRIM(service_type), '')
                AS service_type,

            TRY_TO_DECIMAL(billed_amount::VARCHAR, 12, 2)
                AS billed_amount,

            TRY_TO_DECIMAL(approved_amount::VARCHAR, 12, 2)
                AS approved_amount,

            TRY_TO_DECIMAL(patient_copay::VARCHAR, 12, 2)
                AS patient_copay,

            NULLIF(TRIM(claim_status), '')
                AS claim_status,

            COALESCE(
                TRY_TO_DATE(NULLIF(TRIM(payment_date), ''), 'YYYY-MM-DD'),
                TRY_TO_DATE(NULLIF(TRIM(payment_date), ''), 'DD/MM/YYYY')
            ) AS payment_date,

            NULLIF(TRIM(denial_reason), '')
                AS denial_reason,

            DATEDIFF(
                DAY,
                COALESCE(
                    TRY_TO_DATE(claim_date, 'YYYY-MM-DD'),
                    TRY_TO_DATE(claim_date, 'DD/MM/YYYY')
                ),
                COALESCE(
                    TRY_TO_DATE(NULLIF(TRIM(payment_date), ''), 'YYYY-MM-DD'),
                    TRY_TO_DATE(NULLIF(TRIM(payment_date), ''), 'DD/MM/YYYY')
                )
            ) AS claim_processing_days,

            IFF(
                TRY_TO_DECIMAL(
                    billed_amount::VARCHAR, 12, 2
                ) > 100000,
                TRUE,
                FALSE
            ) AS is_high_value_claim

        FROM RAW.CLAIMS_STREAM

        WHERE METADATA$ACTION = 'INSERT'

        QUALIFY ROW_NUMBER() OVER (
            PARTITION BY claim_id
            ORDER BY METADATA$ROW_ID
        ) = 1

    ) AS src

    ON tgt.claim_id = src.claim_id

    WHEN MATCHED THEN UPDATE SET
        tgt.patient_id = src.patient_id,
        tgt.provider_id = src.provider_id,
        tgt.diagnosis_code = src.diagnosis_code,
        tgt.diagnosis_description = src.diagnosis_description,
        tgt.claim_date = src.claim_date,
        tgt.service_type = src.service_type,
        tgt.billed_amount = src.billed_amount,
        tgt.approved_amount = src.approved_amount,
        tgt.patient_copay = src.patient_copay,
        tgt.claim_status = src.claim_status,
        tgt.payment_date = src.payment_date,
        tgt.denial_reason = src.denial_reason,
        tgt.claim_processing_days = src.claim_processing_days,
        tgt.is_high_value_claim = src.is_high_value_claim

    WHEN NOT MATCHED THEN INSERT (
        claim_id,
        patient_id,
        provider_id,
        diagnosis_code,
        diagnosis_description,
        claim_date,
        service_type,
        billed_amount,
        approved_amount,
        patient_copay,
        claim_status,
        payment_date,
        denial_reason,
        claim_processing_days,
        is_high_value_claim
    )
    VALUES (
        src.claim_id,
        src.patient_id,
        src.provider_id,
        src.diagnosis_code,
        src.diagnosis_description,
        src.claim_date,
        src.service_type,
        src.billed_amount,
        src.approved_amount,
        src.patient_copay,
        src.claim_status,
        src.payment_date,
        src.denial_reason,
        src.claim_processing_days,
        src.is_high_value_claim
    );

    RETURN 'CLAIMS MERGE completed successfully';

END;
$$;

USE DATABASE HEALTHFLOW_DB;
USE WAREHOUSE HF_TRANSFORM_WH;

SELECT
    COUNT(*) AS total_claims,

    COUNT_IF(diagnosis_matches = 0)
        AS claims_without_diagnosis,

    COUNT_IF(diagnosis_matches = 1)
        AS claims_with_one_diagnosis,

    COUNT_IF(diagnosis_matches > 1)
        AS claims_with_multiple_diagnoses

FROM (
    SELECT
        c.claim_id,
        COUNT(d.record_id) AS diagnosis_matches

    FROM CURATED.CLAIMS c

    LEFT JOIN CURATED.DIAGNOSES_HISTORY d
        ON c.patient_id = d.patient_id
        AND c.diagnosis_code = d.diagnosis_code

    GROUP BY c.claim_id
);

CREATE OR REPLACE TABLE CURATED.CLAIM_LINE_ITEM AS

SELECT
    c.claim_id,
    c.patient_id,
    c.provider_id,

    c.diagnosis_code,
    c.diagnosis_description,

    c.claim_date,
    c.service_type,

    c.billed_amount,
    c.approved_amount,
    c.patient_copay,
    c.claim_status,
    c.payment_date,
    c.denial_reason,

    c.claim_processing_days,
    c.is_high_value_claim,

    d.record_id AS diagnosis_record_id,
    d.diagnosis_date,
    d.severity,
    d.is_chronic,
    d.treatment_plan,
    d.follow_up_date,
    d.follow_up_days

FROM CURATED.CLAIMS c

LEFT JOIN CURATED.DIAGNOSES_HISTORY d
    ON c.patient_id = d.patient_id
    AND c.diagnosis_code = d.diagnosis_code
    AND d.diagnosis_date <= c.claim_date

QUALIFY ROW_NUMBER() OVER (
    PARTITION BY c.claim_id

// Date and Business Rule Validation
USE DATABASE HEALTHFLOW_DB;
USE WAREHOUSE HF_TRANSFORM_WH;

SELECT
    COUNT(*) AS total_diagnoses,

    COUNT_IF(diagnosis_date IS NULL)
        AS null_diagnosis_dates,

    COUNT_IF(follow_up_date IS NULL)
        AS null_follow_up_dates,

    COUNT_IF(follow_up_days < 0)
        AS negative_follow_up_days,

    COUNT_IF(
        follow_up_days IS DISTINCT FROM
        DATEDIFF(
            DAY,
            diagnosis_date,
            follow_up_date
        )
    ) AS incorrect_follow_up_days

FROM CURATED.DIAGNOSES_HISTORY;

// Validate insurance claim business rules
SELECT
    COUNT(*) AS total_claims,

    COUNT_IF(claim_date IS NULL)
        AS null_claim_dates,

    COUNT_IF(billed_amount < 0)
        AS negative_billed_amounts,

    COUNT_IF(approved_amount < 0)
        AS negative_approved_amounts,

    COUNT_IF(patient_copay < 0)
        AS negative_copays,

    COUNT_IF(approved_amount > billed_amount)
        AS approved_exceeds_billed,

    COUNT_IF(claim_processing_days < 0)
        AS negative_processing_days,

    COUNT_IF(
        is_high_value_claim IS DISTINCT FROM
        (billed_amount > 100000)
    ) AS incorrect_high_value_flags

FROM CURATED.CLAIMS;

// Check Diagnosis Matching
SELECT
    COUNT(*) AS TOTAL_CLAIMS,

    COUNT_IF(
        EXISTS (
            SELECT 1
            FROM HEALTHFLOW_DB.CURATED.DIAGNOSES_HISTORY d
            WHERE d.PATIENT_ID = c.PATIENT_ID
              AND d.DIAGNOSIS_CODE = c.DIAGNOSIS_CODE
              AND d.DIAGNOSIS_DATE <= c.CLAIM_DATE
        )
    ) AS MATCHED_CLAIMS,

    COUNT_IF(
        NOT EXISTS (
            SELECT 1
            FROM HEALTHFLOW_DB.CURATED.DIAGNOSES_HISTORY d
            WHERE d.PATIENT_ID = c.PATIENT_ID
              AND d.DIAGNOSIS_CODE = c.DIAGNOSIS_CODE
              AND d.DIAGNOSIS_DATE <= c.CLAIM_DATE
        )
    ) AS UNMATCHED_CLAIMS

FROM HEALTHFLOW_DB.CURATED.CLAIMS c;

// Validate Facility and Month Matching
WITH facility_claims AS (
    SELECT
        FACILITY_NAME,
        DATE_TRUNC('MONTH', CLAIM_DATE)::DATE AS CLAIM_MONTH,
        COUNT(*) AS CLAIM_COUNT
    FROM HEALTHFLOW_DB.ANALYTICS.FACT_CLAIMS
    GROUP BY 1, 2
)
SELECT
    COUNT(*) AS TOTAL_FACILITY_MONTH_ROWS,
    COUNT_IF(c.FACILITY_NAME IS NOT NULL)
        AS MATCHED_FACILITY_MONTHS,
    COUNT_IF(c.FACILITY_NAME IS NULL)
        AS UNMATCHED_FACILITY_MONTHS
FROM HEALTHFLOW_DB.CURATED.FACILITY_MONTHLY_METRICS f
LEFT JOIN facility_claims c
    ON TRIM(UPPER(f.FACILITY_NAME))
       = TRIM(UPPER(c.FACILITY_NAME))
   AND DATE_TRUNC('MONTH', f.REPORT_MONTH)::DATE
       = c.CLAIM_MONTH;

// Claim Reconciliation
WITH facility_months AS (
    SELECT DISTINCT
        TRIM(UPPER(FACILITY_NAME)) AS FACILITY_KEY,
        DATE_TRUNC('MONTH', REPORT_MONTH)::DATE AS REPORT_MONTH
    FROM HEALTHFLOW_DB.CURATED.FACILITY_MONTHLY_METRICS
)
SELECT
    COUNT(*) AS TOTAL_CLAIMS,

    COUNT_IF(f.FACILITY_KEY IS NOT NULL)
        AS MATCHED_CLAIMS,

    COUNT_IF(f.FACILITY_KEY IS NULL)
        AS UNMATCHED_CLAIMS

FROM HEALTHFLOW_DB.ANALYTICS.FACT_CLAIMS c

LEFT JOIN facility_months f
    ON TRIM(UPPER(c.FACILITY_NAME)) = f.FACILITY_KEY
   AND DATE_TRUNC('MONTH', c.CLAIM_DATE)::DATE
       = f.REPORT_MONTH;

// Month-over-Month Revenue Growth
WITH monthly_revenue AS (
    SELECT
        DATE_TRUNC('MONTH', REPORT_MONTH)::DATE AS MONTH,
        SUM(TOTAL_REVENUE) AS REVENUE
    FROM HEALTHFLOW_DB.ANALYTICS.MONTHLY_FACILITY_DASHBOARD
    GROUP BY 1
),

revenue_comparison AS (
    SELECT
        MONTH,
        REVENUE,
        LAG(REVENUE) OVER (
            ORDER BY MONTH
        ) AS PREVIOUS_MONTH_REVENUE
    FROM monthly_revenue
)

SELECT
    MONTH,
    REVENUE,
    PREVIOUS_MONTH_REVENUE,

    ROUND(
        (REVENUE - PREVIOUS_MONTH_REVENUE)
        * 100.0 /
        NULLIF(PREVIOUS_MONTH_REVENUE, 0),
        2
    ) AS MOM_GROWTH_PCT

FROM revenue_comparison
ORDER BY MONTH;


// Running Total of Claims per Patient
SELECT
    PATIENT_ID,
    CLAIM_ID,
    CLAIM_DATE,
    CLAIM_STATUS,
    BILLED_AMOUNT,

    COUNT(*) OVER (
        PARTITION BY PATIENT_ID
        ORDER BY CLAIM_DATE, CLAIM_ID
        ROWS BETWEEN UNBOUNDED PRECEDING
        AND CURRENT ROW
    ) AS RUNNING_CLAIM_COUNT

FROM HEALTHFLOW_DB.ANALYTICS.FACT_CLAIMS

ORDER BY PATIENT_ID, CLAIM_DATE, CLAIM_ID
LIMIT 30;

// Top 3 Diagnoses per Region
WITH diagnosis_counts AS (
    SELECT
        PATIENT_REGION,
        DIAGNOSIS_CODE,
        COUNT(*) AS TOTAL_CLAIMS

    FROM HEALTHFLOW_DB.ANALYTICS.FACT_CLAIMS

    WHERE PATIENT_REGION IS NOT NULL
      AND DIAGNOSIS_CODE IS NOT NULL

    GROUP BY
        PATIENT_REGION,
        DIAGNOSIS_CODE
)

SELECT
    PATIENT_REGION,
    DIAGNOSIS_CODE,
    TOTAL_CLAIMS,

    ROW_NUMBER() OVER (
        PARTITION BY PATIENT_REGION
        ORDER BY TOTAL_CLAIMS DESC, DIAGNOSIS_CODE
    ) AS DIAGNOSIS_RANK

FROM diagnosis_counts

QUALIFY DIAGNOSIS_RANK <= 3

ORDER BY PATIENT_REGION, DIAGNOSIS_RANK;

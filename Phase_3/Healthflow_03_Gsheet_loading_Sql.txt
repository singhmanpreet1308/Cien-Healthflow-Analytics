USE DATABASE HEALTHFLOW_DB;
USE SCHEMA RAW;

SELECT
    'MEDICATIONS_REFERENCE' AS TABLE_NAME,
    COUNT(*) AS TOTAL_ROWS
FROM MEDICATIONS_REFERENCE

UNION ALL

SELECT
    'FACILITY_MONTHLY_METRICS',
    COUNT(*)
FROM FACILITY_MONTHLY_METRICS;

// Check duplicate IDs
SELECT
    medication_id,
    COUNT(*) AS duplicate_count
FROM RAW.MEDICATIONS_REFERENCE
GROUP BY medication_id
HAVING COUNT(*) > 1;

SELECT
    metric_id,
    COUNT(*) AS duplicate_count
FROM RAW.FACILITY_MONTHLY_METRICS
GROUP BY metric_id
HAVING COUNT(*) > 1;

// Verify NULL values
SELECT
    COUNT_IF(medication_id IS NULL) AS NULL_IDS,
    COUNT_IF(medication_name IS NULL) AS NULL_NAMES
FROM RAW.MEDICATIONS_REFERENCE;

SELECT
    COUNT_IF(metric_id IS NULL) AS NULL_IDS,
    COUNT_IF(facility_name IS NULL) AS NULL_FACILITIES,
    COUNT_IF(report_month IS NULL) AS NULL_MONTHS
FROM RAW.FACILITY_MONTHLY_METRICS;

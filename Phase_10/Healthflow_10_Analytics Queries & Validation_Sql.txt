USE DATABASE HEALTHFLOW_DB;
USE SCHEMA ANALYTICS;
USE WAREHOUSE HF_ANALYTICS_WH;

// Verify table columns
DESC TABLE HEALTHFLOW_DB.CURATED.CLAIMS;

// Query 1- Claims Analysis SQL
SELECT
    p.INSURANCE_PROVIDER,
    c.SERVICE_TYPE,
    COUNT(*) AS TOTAL_CLAIMS,

    ROUND(
        100.0 * COUNT_IF(
            UPPER(c.CLAIM_STATUS) = 'APPROVED'
        ) / NULLIF(COUNT(*), 0), 2
    ) AS APPROVAL_RATE_PCT,

    ROUND(AVG(c.BILLED_AMOUNT), 2)
        AS AVG_BILLED_AMOUNT,

    ROUND(SUM(
        CASE
            WHEN UPPER(c.CLAIM_STATUS) = 'APPROVED'
            THEN COALESCE(c.APPROVED_AMOUNT, 0)
            ELSE 0
        END
    ), 2) AS TOTAL_APPROVED_AMOUNT,

    ROUND(
        100.0 * COUNT_IF(
            UPPER(c.CLAIM_STATUS) = 'DENIED'
        ) / NULLIF(COUNT(*), 0), 2
    ) AS DENIAL_RATE_PCT

FROM HEALTHFLOW_DB.CURATED.CLAIMS c

LEFT JOIN HEALTHFLOW_DB.CURATED.PATIENTS p
    ON c.PATIENT_ID = p.PATIENT_ID

GROUP BY
    p.INSURANCE_PROVIDER,
    c.SERVICE_TYPE

ORDER BY DENIAL_RATE_PCT DESC;

// Query 2 — Chronic Disease, Chronic vs non_chronic, Avg_claim
DESC TABLE HEALTHFLOW_DB.CURATED.PATIENTS;

DESC TABLE HEALTHFLOW_DB.CURATED.DIAGNOSES_HISTORY;

-- 1)
SELECT
    DIAGNOSIS_CODE,
    DIAGNOSIS_DESCRIPTION,
    COUNT(DISTINCT PATIENT_ID) AS TOTAL_PATIENTS

FROM HEALTHFLOW_DB.CURATED.DIAGNOSES_HISTORY

WHERE IS_CHRONIC = TRUE

GROUP BY
    DIAGNOSIS_CODE,
    DIAGNOSIS_DESCRIPTION

ORDER BY TOTAL_PATIENTS DESC
LIMIT 5;

-- 2)
WITH patient_status AS (
    SELECT
        p.PATIENT_ID,
        MAX(
            IFF(d.IS_CHRONIC = TRUE, 1, 0)
        ) AS HAS_CHRONIC
    FROM HEALTHFLOW_DB.CURATED.PATIENTS p
    LEFT JOIN HEALTHFLOW_DB.CURATED.DIAGNOSES_HISTORY d
        ON p.PATIENT_ID = d.PATIENT_ID
    GROUP BY p.PATIENT_ID
),

claim_summary AS (
    SELECT
        PATIENT_ID,
        SUM(BILLED_AMOUNT) AS TOTAL_BILLED,
        COUNT(*) AS CLAIM_COUNT
    FROM HEALTHFLOW_DB.CURATED.CLAIMS
    GROUP BY PATIENT_ID
)

SELECT
    IFF(ps.HAS_CHRONIC = 1, 'Chronic', 'Non-Chronic')
        AS PATIENT_CATEGORY,

    COUNT(*) AS TOTAL_PATIENTS,

    ROUND(
        SUM(COALESCE(cs.TOTAL_BILLED, 0))
        / NULLIF(SUM(COALESCE(cs.CLAIM_COUNT, 0)), 0),
        2
    ) AS AVG_CLAIM_AMOUNT

FROM patient_status ps
LEFT JOIN claim_summary cs
    ON ps.PATIENT_ID = cs.PATIENT_ID

GROUP BY ps.HAS_CHRONIC
ORDER BY ps.HAS_CHRONIC DESC;


// Query 3: Provider Performance
DESC TABLE HEALTHFLOW_DB.CURATED.PROVIDERS;

WITH provider_performance AS (
    SELECT
        p.PROVIDER_ID,
        p.PROVIDER_NAME,
        p.SPECIALIZATION,
        COUNT(c.CLAIM_ID) AS TOTAL_CLAIMS,

        ROUND(SUM(
            CASE
                WHEN UPPER(c.CLAIM_STATUS) = 'APPROVED'
                THEN COALESCE(c.APPROVED_AMOUNT, 0)
                ELSE 0
            END
        ), 2) AS TOTAL_APPROVED_AMOUNT

    FROM HEALTHFLOW_DB.CURATED.PROVIDERS p

    LEFT JOIN HEALTHFLOW_DB.CURATED.CLAIMS c
        ON p.PROVIDER_ID = c.PROVIDER_ID

    GROUP BY
        p.PROVIDER_ID,
        p.PROVIDER_NAME,
        p.SPECIALIZATION
)

SELECT
    *,
    DENSE_RANK() OVER (
        PARTITION BY SPECIALIZATION
        ORDER BY TOTAL_APPROVED_AMOUNT DESC
    ) AS PROVIDER_RANK

FROM provider_performance

QUALIFY PROVIDER_RANK <= 5

ORDER BY SPECIALIZATION, PROVIDER_RANK;

// Query 4 — Facility Benchmarking
DESC TABLE HEALTHFLOW_DB.ANALYTICS.MONTHLY_FACILITY_DASHBOARD;


SELECT
    FACILITY_NAME,

    ROUND(AVG(PATIENT_SATISFACTION_SCORE), 2)
        AS AVG_SATISFACTION_SCORE,

    ROUND(AVG(AVG_WAIT_TIME_MINUTES), 2)
        AS AVG_WAIT_TIME_MINUTES,

    CASE
        WHEN AVG(PATIENT_SATISFACTION_SCORE) < 3.0
         AND AVG(AVG_WAIT_TIME_MINUTES) > 60
        THEN 'Below Benchmark'
        ELSE 'Meets Benchmark'
    END AS BENCHMARK_STATUS

FROM HEALTHFLOW_DB.ANALYTICS.MONTHLY_FACILITY_DASHBOARD

GROUP BY FACILITY_NAME

ORDER BY
    AVG_SATISFACTION_SCORE ASC,
    AVG_WAIT_TIME_MINUTES DESC;


// Query 5 — Revenue Trends
DESC TABLE HEALTHFLOW_DB.ANALYTICS.MONTHLY_FACILITY_DASHBOARD;
DESC TABLE HEALTHFLOW_DB.ANALYTICS.DIM_PROVIDERS;


WITH monthly_revenue AS (
    SELECT
        DATE_TRUNC('MONTH', REPORT_MONTH) AS REPORT_MONTH,
        FACILITY_NAME,
        SUM(TOTAL_REVENUE) AS MONTHLY_REVENUE
    FROM HEALTHFLOW_DB.CURATED.FACILITY_MONTHLY_METRICS
    GROUP BY 1, 2
),

revenue_trends AS (
    SELECT
        REPORT_MONTH,
        FACILITY_NAME,
        MONTHLY_REVENUE,

        LAG(MONTHLY_REVENUE) OVER (
            PARTITION BY FACILITY_NAME
            ORDER BY REPORT_MONTH
        ) AS PREVIOUS_MONTH_REVENUE,

        SUM(MONTHLY_REVENUE) OVER (
            PARTITION BY FACILITY_NAME
            ORDER BY REPORT_MONTH
            ROWS BETWEEN UNBOUNDED PRECEDING
                     AND CURRENT ROW
        ) AS RUNNING_REVENUE

    FROM monthly_revenue
)

SELECT
    REPORT_MONTH,
    FACILITY_NAME,
    ROUND(MONTHLY_REVENUE, 2) AS MONTHLY_REVENUE,

    ROUND(
        100.0 * (MONTHLY_REVENUE - PREVIOUS_MONTH_REVENUE)
        / NULLIF(PREVIOUS_MONTH_REVENUE, 0), 2
    ) AS MOM_GROWTH_PCT,

    ROUND(RUNNING_REVENUE, 2) AS RUNNING_REVENUE

FROM revenue_trends
ORDER BY FACILITY_NAME, REPORT_MONTH;

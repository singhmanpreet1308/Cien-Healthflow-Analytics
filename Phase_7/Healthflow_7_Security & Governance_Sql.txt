-- Email masking policy
CREATE MASKING POLICY IF NOT EXISTS MASK_EMAIL
AS (VAL STRING) RETURNS STRING ->
    CASE
        WHEN CURRENT_ROLE() = 'HEALTHFLOW_ADMIN'
            THEN VAL
        WHEN VAL IS NULL
            THEN NULL
        ELSE '********@masked.com'
    END;

-- Phone masking policy

AS (VAL STRING) RETURNS STRING ->
    CASE
        WHEN CURRENT_ROLE() = 'HEALTHFLOW_ADMIN'
            THEN VAL
        WHEN VAL IS NULL
            THEN NULL
        ELSE 'XXX-XXX-XXXX'
    END;

HEALTHFLOW_DB.GOVERNANCE.ROLE_REGION_MAP (
    ROLE_NAME VARCHAR(100),
    REGION VARCHAR(100)

// Create the Row Access Policy
USE ROLE ACCOUNTADMIN;
USE WAREHOUSE HF_DEV_WH;

CREATE ROW ACCESS POLICY IF NOT EXISTS
HEALTHFLOW_DB.GOVERNANCE.CLAIMS_REGION_POLICY
AS (P_CLAIM_PATIENT_ID NUMBER)
RETURNS BOOLEAN ->
    CURRENT_ROLE() = 'HEALTHFLOW_ADMIN'
    OR EXISTS (
        SELECT 1
        FROM HEALTHFLOW_DB.CURATED.PATIENTS P
        JOIN HEALTHFLOW_DB.GOVERNANCE.ROLE_REGION_MAP R
          ON P.ADDRESS_STATE = R.REGION
        WHERE P.PATIENT_ID = P_CLAIM_PATIENT_ID
          AND R.ROLE_NAME = CURRENT_ROLE()
    );

// Verify the policy
SHOW ROW ACCESS POLICIES
IN SCHEMA HEALTHFLOW_DB.GOVERNANCE;


// Test Row-Level Security
USE ROLE ACCOUNTADMIN;
USE WAREHOUSE HF_DEV_WH;

CREATE OR REPLACE TABLE
HEALTHFLOW_DB.GOVERNANCE.CLAIMS_RAP_TEST AS
SELECT
    C.CLAIM_ID,
    C.PATIENT_ID,
    P.ADDRESS_STATE AS REGION
FROM HEALTHFLOW_DB.CURATED.CLAIMS C
JOIN HEALTHFLOW_DB.CURATED.PATIENTS P
    ON C.PATIENT_ID = P.PATIENT_ID
WHERE P.ADDRESS_STATE IN (
    'Maharashtra',
    'Tamil Nadu',
    'Kerala'
)
LIMIT 100;

// Attach the Row Access Policy
ALTER TABLE HEALTHFLOW_DB.GOVERNANCE.CLAIMS_RAP_TEST
ADD ROW ACCESS POLICY
HEALTHFLOW_DB.GOVERNANCE.CLAIMS_REGION_POLICY
ON (PATIENT_ID);
